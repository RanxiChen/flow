cmake_minimum_required(VERSION 3.16)
project(r01_runner LANGUAGES C CXX)
if(NOT R01_LLAMA_SOURCE)
    message(FATAL_ERROR "Set R01_LLAMA_SOURCE to the pinned upstream checkout")
endif()
add_subdirectory(${R01_LLAMA_SOURCE} llama-build)
add_executable(r01-runner main.cpp)
target_link_libraries(r01-runner PRIVATE llama)
target_compile_features(r01-runner PRIVATE cxx_std_17)
target_compile_options(r01-runner PRIVATE -g)
